CREATE DATABASE peluqueria;

\c peluqueria;

-- CLIENTES
CREATE TABLE clientes (
    id_clientes INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    telefono VARCHAR(20),
    email VARCHAR(100) UNIQUE
);

-- EMPLEADOS
CREATE TABLE empleados (
    id_empleados INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL
);

-- SERVICIOS
CREATE TABLE servicios (
    id_servicios INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    precio NUMERIC(10,2) NOT NULL CHECK (precio >= 0),
    duracion INT NOT NULL CHECK (duracion > 0)
);

-- TURNOS
CREATE TABLE turnos (
    id_turnos INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    cliente_id INT NOT NULL REFERENCES clientes(id_clientes),
    empleado_id INT NOT NULL REFERENCES empleados(id_empleados),
    servicio_id INT NOT NULL REFERENCES servicios(id_servicios),
    fecha TIMESTAMP NOT NULL,
    estado VARCHAR(20) NOT NULL DEFAULT 'pendiente',
    
    CHECK (estado IN ('pendiente', 'confirmado', 'cancelado', 'finalizado'))
);

-- CLIENTES
CREATE TABLE clientes (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    telefono VARCHAR(20),
    email VARCHAR(100)
);

-- EMPLEADOS (peluqueros)
CREATE TABLE empleados (
    id SERIAL PRIMARY KEY,

);

-- SERVICIOS
CREATE TABLE servicios (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    precio NUMERIC(10,2) NOT NULL,
    duracion INT NOT NULL -- minutos
);

-- TURNOS
CREATE TABLE turnos (
    id SERIAL PRIMARY KEY,
    cliente_id INT REFERENCES clientes(id),
    empleado_id INT REFERENCES empleados(id),
    servicio_id INT REFERENCES servicios(id),
    fecha TIMESTAMP NOT NULL,
    estado VARCHAR(20) DEFAULT 'pendiente'
);
